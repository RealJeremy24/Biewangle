## 2. 设置闹钟 — `AlarmScheduler.kt`

> 核心思路：用 `PendingIntent.getActivity()` 直接启动 `AlarmActivity`，模仿系统时钟 App。用 `setAlarmClock()` 获得最高闹钟优先级（状态栏显示闹钟图标，国产 ROM 最尊重）。

```kotlin
class AlarmScheduler(private val context: Context) {

    fun schedule(
        memoId: Long, title: String, content: String,
        alarmTime: Long, useCustomRingtone: Boolean = true
    ) {
        // 闹钟到点 → 直接打开 AlarmActivity（不走 BroadcastReceiver）
        val intent = Intent(context, AlarmActivity::class.java).apply {
            action = Constants.ACTION_SHOW_REMINDER
            putExtra(Constants.EXTRA_MEMO_ID, memoId)
            putExtra(Constants.EXTRA_MEMO_TITLE, title)
            putExtra(Constants.EXTRA_MEMO_CONTENT, content)
            putExtra(Constants.EXTRA_USE_CUSTOM_RINGTONE, useCustomRingtone)
            flags = Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TOP
        }

        val requestCode = memoId.toInt()
        val flags = PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        val pendingIntent = PendingIntent.getActivity(context, requestCode, intent, flags)

        val alarmManager = context.getSystemService(Context.ALARM_SERVICE) as AlarmManager

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
            alarmManager.setAlarmClock(
                AlarmManager.AlarmClockInfo(alarmTime, pendingIntent),
                pendingIntent
            )
        } else {
            @Suppress("DEPRECATION")
            alarmManager.setExact(AlarmManager.RTC_WAKEUP, alarmTime, pendingIntent)
        }
    }

    fun cancel(memoId: Long) {
        val intent = Intent(context, AlarmReceiver::class.java).apply {
            action = Constants.ACTION_SHOW_REMINDER
        }
        val pendingIntent = PendingIntent.getBroadcast(
            context, memoId.toInt(), intent,
            PendingIntent.FLAG_NO_CREATE or PendingIntent.FLAG_IMMUTABLE
        )
        pendingIntent?.let {
            val alarmManager = context.getSystemService(Context.ALARM_SERVICE) as AlarmManager
            alarmManager.cancel(it)
            it.cancel()
        }
    }
}
```

**⚠️ 已知问题：`cancel()` 中使用的是旧的 `getBroadcast(AlarmReceiver)`，与 `schedule()` 中新的 `getActivity(AlarmActivity)` 的 PendingIntent 不匹配，cancel 实际上无效。**

## 3. BroadcastReceiver — `AlarmReceiver.kt`

> 新闹钟已不走这里（直接用 `getActivity`），此 Receiver 仅作为旧闹钟兼容路径和 dismiss/snooze 按钮兼容。

```kotlin
class AlarmReceiver : BroadcastReceiver() {

    override fun onReceive(context: Context, intent: Intent) {
        val memoId = intent.getLongExtra(Constants.EXTRA_MEMO_ID, -1L)
        if (memoId == -1L) return

        when (intent.action) {
            Constants.ACTION_DISMISS -> {
                AlarmForegroundService.stop(context)
                val nm = context.getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
                nm.cancel(memoId.toInt())
            }
            Constants.ACTION_SNOOZE -> {
                AlarmForegroundService.stop(context)
                val nm = context.getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
                nm.cancel(memoId.toInt())

                val title = intent.getStringExtra(Constants.EXTRA_MEMO_TITLE) ?: "提醒"
                val content = intent.getStringExtra(Constants.EXTRA_MEMO_CONTENT) ?: ""
                val snoozeTime = System.currentTimeMillis() + Constants.SNOOZE_DELAY_MS
                AlarmScheduler(context).schedule(memoId, title, content, snoozeTime, true)
            }
            else -> {
                // 旧闹钟兼容：BroadcastReceiver → 手动打开 AlarmActivity
                val title = intent.getStringExtra(Constants.EXTRA_MEMO_TITLE) ?: "提醒"
                val content = intent.getStringExtra(Constants.EXTRA_MEMO_CONTENT) ?: ""
                val useCustomRingtone = intent.getBooleanExtra(Constants.EXTRA_USE_CUSTOM_RINGTONE, true)

                val alarmIntent = Intent(context, AlarmActivity::class.java).apply {
                    action = Constants.ACTION_SHOW_REMINDER
                    putExtra(Constants.EXTRA_MEMO_ID, memoId)
                    putExtra(Constants.EXTRA_MEMO_TITLE, title)
                    putExtra(Constants.EXTRA_MEMO_CONTENT, content)
                    putExtra(Constants.EXTRA_USE_CUSTOM_RINGTONE, useCustomRingtone)
                    flags = Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TOP
                }
                context.startActivity(alarmIntent)
            }
        }
    }
}
```

## 4. 全屏提醒 + 通知 — `AlarmActivity.kt`

> 整个闹钟机制的核心。闹钟到点后系统直接拉起这个 Activity，**在 Activity 内直接播放铃声**（而不是依赖前台服务），同时发一条状态栏通知作为伴侣（用户按 Home 后可通过通知返回或关闭）。

### 4.1 Window 配置 & Intent 路由

```kotlin
class AlarmActivity : ComponentActivity() {

    private var memoId: Long = -1L
    private var title: String = ""
    private var content: String = ""
    private var useCustomRingtone: Boolean = true
    private var player: ReminderPlayer? = null

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        // 全屏 + 点亮屏幕 + 覆盖锁屏
        window.apply {
            addFlags(WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON)
            addFlags(WindowManager.LayoutParams.FLAG_SHOW_WHEN_LOCKED)
            addFlags(WindowManager.LayoutParams.FLAG_TURN_SCREEN_ON)
            addFlags(WindowManager.LayoutParams.FLAG_DISMISS_KEYGUARD)
        }

        handleIntent(intent)
    }

    override fun onNewIntent(intent: Intent) {
        super.onNewIntent(intent)
        handleIntent(intent)
    }

    private fun handleIntent(intent: Intent?) {
        if (intent == null) return
        memoId = intent.getLongExtra(Constants.EXTRA_MEMO_ID, -1L)
        title = intent.getStringExtra(Constants.EXTRA_MEMO_TITLE) ?: "提醒"
        content = intent.getStringExtra(Constants.EXTRA_MEMO_CONTENT) ?: ""
        useCustomRingtone = intent.getBooleanExtra(Constants.EXTRA_USE_CUSTOM_RINGTONE, true)

        when (intent.action) {
            Constants.ACTION_SHOW_REMINDER -> startAlarm()
            Constants.ACTION_DISMISS -> dismissAlarm()
            Constants.ACTION_SNOOZE -> snoozeAlarm()
        }
    }
```

### 4.2 启动闹钟（铃声 + 通知）

```kotlin
    private fun startAlarm() {
        // 直接播放铃声 —— ReminderPlayer 使用 MediaPlayer，
        // 支持自定义铃声、裁剪片段循环、震动
        if (player == null) {
            player = ReminderPlayer(this)
            player?.startLooping(useCustomRingtone)
        }

        // 伴侣通知（用户按 Home 后仍能看到）
        createNotification()

        setContent {
            AlarmScreen(
                title = title,
                content = content,
                onDismiss = { dismissAlarm() },
                onSnooze = { snoozeAlarm() }
            )
        }
    }
```

### 4.3 通知构建

```kotlin
    private fun createNotification() {
        // 「关闭」→ AlarmActivity（ACTION_DISMISS）
        val dismissIntent = Intent(this, AlarmActivity::class.java).apply {
            action = Constants.ACTION_DISMISS
            putExtra(Constants.EXTRA_MEMO_ID, memoId)
            flags = Intent.FLAG_ACTIVITY_NEW_TASK or
                    Intent.FLAG_ACTIVITY_CLEAR_TOP or
                    Intent.FLAG_ACTIVITY_SINGLE_TOP
        }
        val dismissPI = PendingIntent.getActivity(
            this, memoId.toInt() + 10000, dismissIntent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        // 「稍后提醒」→ AlarmActivity（ACTION_SNOOZE）
        val snoozeIntent = Intent(this, AlarmActivity::class.java).apply {
            action = Constants.ACTION_SNOOZE
            putExtra(Constants.EXTRA_MEMO_ID, memoId)
            putExtra(Constants.EXTRA_MEMO_TITLE, title)
            putExtra(Constants.EXTRA_MEMO_CONTENT, content)
            flags = Intent.FLAG_ACTIVITY_NEW_TASK or
                    Intent.FLAG_ACTIVITY_CLEAR_TOP or
                    Intent.FLAG_ACTIVITY_SINGLE_TOP
        }
        val snoozePI = PendingIntent.getActivity(
            this, memoId.toInt() + 20000, snoozeIntent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val notification = NotificationCompat.Builder(this, Constants.CHANNEL_REMINDER)
            .setSmallIcon(R.drawable.ic_notification)
            .setContentTitle("🔔 $title")
            .setContentText(content.ifEmpty { "提醒时间到" })
            .setPriority(NotificationCompat.PRIORITY_MAX)
            .setCategory(NotificationCompat.CATEGORY_ALARM)
            .setOngoing(true)
            .setAutoCancel(false)
            .addAction(android.R.drawable.ic_menu_close_clear_cancel, "关闭", dismissPI)
            .addAction(android.R.drawable.ic_media_pause, "稍后提醒", snoozePI)
            .build()

        NotificationManagerCompat.from(this).notify(memoId.toInt(), notification)
    }
```

### 4.4 「我知道了」（关闭 + 标记完成）

```kotlin
    private fun dismissAlarm() {
        player?.stop()
        player = null

        val nm = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        nm.cancel(memoId.toInt())

        // 自动标记完成
        CoroutineScope(Dispatchers.IO).launch {
            val memo = BiewangleApp.instance.memoRepository.getById(memoId)
            if (memo != null) {
                BiewangleApp.instance.memoRepository.toggleComplete(memoId)
            }
        }

        finish()
    }
```

### 4.5 「稍后提醒」（10 分钟后重新闹钟）

```kotlin
    private fun snoozeAlarm() {
        player?.stop()
        player = null

        val nm = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        nm.cancel(memoId.toInt())

        val snoozeTime = System.currentTimeMillis() + Constants.SNOOZE_DELAY_MS
        AlarmScheduler(this).schedule(memoId, title, content, snoozeTime, useCustomRingtone)

        finish()
    }
}
```

## 5. 通知渠道 — `BiewangleApp.kt`

```kotlin
private fun createNotificationChannels() {
    if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
        val channels = listOf(
            NotificationChannel(
                Constants.CHANNEL_REMINDER,        // "reminder_channel"
                "提醒通知",                          // 用户可见名称
                NotificationManager.IMPORTANCE_HIGH  // 弹出 + 声音
            ).apply {
                description = "用于显示提醒通知"
                setShowBadge(true)
                enableVibration(true)
                setBypassDnd(true)                  // 免打扰模式下也响
            },
            NotificationChannel(
                Constants.CHANNEL_SERVICE,           // "reminder_service_channel"
                "提醒服务",
                NotificationManager.IMPORTANCE_LOW    // 不弹出，仅在通知栏
            ).apply {
                description = "提醒前台服务运行中"
                setShowBadge(false)
            }
        )

        val manager = getSystemService(NotificationManager::class.java)
        channels.forEach { manager.createNotificationChannel(it) }
    }
}
```

> ⚠️ 注意：`CHANNEL_REMINDER` 没有显式设置 `setSound()`，通知声音依赖系统默认。如果用户选择了自定义铃声，当前实现只在 `ReminderPlayer`（MediaPlayer）中播放，**通知本身不会播放自定义铃声**。这意味着如果 `AlarmActivity` 没能弹出（例如国产 ROM 限制），用户可能收到的是静默通知或系统默认提示音。

## 当前架构的数据流

```
用户创建提醒
  → AlarmScheduler.schedule()
    → AlarmManager.setAlarmClock(
        PendingIntent.getActivity(AlarmActivity)  // 闹钟到点 → 系统直接启动 Activity
      )

闹钟到点
  → AlarmActivity.onCreate()
    ├── window flags：KEEP_SCREEN_ON + SHOW_WHEN_LOCKED + TURN_SCREEN_ON + DISMISS_KEYGUARD
    ├── ReminderPlayer.startLooping()   // MediaPlayer 直接播铃声 + 震动
    ├── createNotification()            // 状态栏伴侣通知
    └── setContent(AlarmScreen)         // 全屏 UI

用户操作
  ├── 「我知道了」→ dismissAlarm() → player.stop() + 取消通知 + 标记完成 + finish()
  └── 「稍后提醒」→ snoozeAlarm()  → player.stop() + 取消通知 + 重新调度 + finish()

用户按 Home（闹钟仍在响）
  → 通知栏有 ongoing 通知
    ├── 「关闭」→ pendingIntent → AlarmActivity(ACTION_DISMISS) → onNewIntent → dismissAlarm()
    └── 「稍后提醒」→ pendingIntent → AlarmActivity(ACTION_SNOOZE) → onNewIntent → snoozeAlarm()
```

## 已知问题

| # | 问题 | 影响 |
|---|------|------|
| 1 | `AlarmScheduler.cancel()` 的 PendingIntent 类型与 `schedule()` 不匹配（getBroadcast vs getActivity），取消功能**实际无效** | 闹钟一旦设置无法取消 |
| 2 | 通知 (`createNotification()`) 没有 `.setSound()` 和 `.setFullScreenIntent()`，声音完全依赖 `ReminderPlayer` | 如果 AlarmActivity 因 ROM 限制未能弹出，用户收到静默通知 |
| 3 | Android 14+ 的 `USE_FULL_SCREEN_INTENT` 权限虽已声明，但通知未使用 `setFullScreenIntent()` | 声明了但没用到 |
| 4 | `FLAG_SHOW_WHEN_LOCKED`、`FLAG_TURN_SCREEN_ON`、`FLAG_DISMISS_KEYGUARD` 在 API 27+ 已 deprecated，应改用 `R.attr.showWhenLocked` 等或 `setTurnScreenOn(true)` / `setShowWhenLocked(true)` | 在最新系统上可能行为不一致 |
| 5 | 国产 ROM（小米/华为/OPPO/vivo）在 App 被划掉后可能冻结进程，即使 `setAlarmClock` 也可能被延迟 | 取决于用户在系统「自启动」设置中是否允许 |
